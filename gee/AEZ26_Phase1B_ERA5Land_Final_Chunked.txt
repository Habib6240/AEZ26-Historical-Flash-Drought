// =====================================================================
// 3. HOURLY PHYSICS
// =====================================================================

/*
 * LEARNING FLOW
 *
 * ERA5 hourly raster
 *       ↓
 * Temperature K -> °C
 *       ↓
 * Dewpoint K -> °C
 *       ↓
 * hourly vapor-pressure physics
 *       ↓
 * hourly RH
 * hourly VPD
 *       ↓
 * hourly wind magnitude
 *
 * Only AFTER these calculations do we aggregate to daily values.
 */


function processERA5Hourly(img) {

  // ---------------------------------------------------------------
  // A. TEMPERATURE
  // ---------------------------------------------------------------

  /*
   * ERA5 temperature units:
   *
   * Kelvin
   *
   * Celsius =
   * Kelvin - 273.15
   */

  var temperatureC =
    img
      .select(
        'temperature_2m'
      )
      .subtract(
        273.15
      );


  var dewpointC =
    img
      .select(
        'dewpoint_temperature_2m'
      )
      .subtract(
        273.15
      );


  // ---------------------------------------------------------------
  // B. WIND SPEED
  // ---------------------------------------------------------------

  /*
   * ERA5 gives wind as vector components:
   *
   * u = east-west component
   * v = north-south component
   *
   * Wind speed:
   *
   * sqrt(u² + v²)
   *
   * IMPORTANT:
   *
   * We calculate hourly wind magnitude FIRST.
   *
   * We do NOT calculate:
   *
   * sqrt(mean(u)² + mean(v)²)
   *
   * because changing wind direction can make components cancel.
   */

  var u10 =
    img.select(
      'u_component_of_wind_10m'
    );

  var v10 =
    img.select(
      'v_component_of_wind_10m'
    );


  var windSpeed =
    u10
      .pow(2)
      .add(
        v10.pow(2)
      )
      .sqrt()
      .rename(
        'wind_speed_m_s'
      );


  // ---------------------------------------------------------------
  // C. SATURATION VAPOR PRESSURE
  // ---------------------------------------------------------------

  /*
   * Tetens-type equation:
   *
   * es =
   * 0.61078 × exp[
   *   (17.27 × T) /
   *   (T + 237.3)
   * ]
   *
   * Unit = kPa
   */

  var saturationVaporPressure =
    temperatureC.expression(

      '0.61078 * exp((17.27 * T) / (T + 237.3))',

      {
        T: temperatureC
      }

    );


  // ---------------------------------------------------------------
  // D. ACTUAL VAPOR PRESSURE
  // ---------------------------------------------------------------

  /*
   * Dewpoint temperature represents saturation
   * temperature at the existing water-vapor content.
   *
   * Therefore actual vapor pressure is estimated
   * from dewpoint using the same equation.
   */

  var actualVaporPressure =
    dewpointC.expression(

      '0.61078 * exp((17.27 * Td) / (Td + 237.3))',

      {
        Td: dewpointC
      }

    );


  // ---------------------------------------------------------------
  // E. VAPOR PRESSURE DEFICIT
  // ---------------------------------------------------------------

  /*
   * VPD =
   * saturation vapor pressure
   * -
   * actual vapor pressure
   *
   * Unit = kPa
   */

  var VPD =
    saturationVaporPressure
      .subtract(
        actualVaporPressure
      )
      .max(0)
      .rename(
        'vpd_kPa'
      );


  // ---------------------------------------------------------------
  // F. RELATIVE HUMIDITY
  // ---------------------------------------------------------------

  /*
   * RH =
   *
   * actual vapor pressure
   * ---------------------
   * saturation vapor pressure
   *
   * × 100
   */

  var RH =
    actualVaporPressure
      .divide(
        saturationVaporPressure
      )
      .multiply(
        100
      )
      .clamp(
        0,
        100
      )
      .rename(
        'rh_pct'
      );


  return ee.Image.cat([

    temperatureC.rename(
      't2m_C'
    ),

    dewpointC.rename(
      'td_C'
    ),

    RH,

    VPD,

    windSpeed

  ])
    .copyProperties(
      img,
      [
        'system:time_start'
      ]
    );

}


// Apply the same physics to every hourly ERA5 image.
var era5HourlyProcessed =
  era5HourlyCol.map(
    processERA5Hourly
  );


// =====================================================================
// 4. DAILY FEATURE-COLLECTION BUILDER
// =====================================================================

/*
 * THIS FUNCTION IS THE CORE OF PHASE 1B.
 *
 * For any requested time period:
 *
 * hourly ERA5
 *      ↓
 * temporal daily aggregation
 *      ↓
 * daily raster
 *      ↓
 * spatial AEZ mean
 *      ↓
 * ee.Feature
 *      ↓
 * one CSV row per day
 */


function buildERA5DailyFC(
  startString,
  endString
) {

  var startDate =
    ee.Date(
      startString
    );

  var endDate =
    ee.Date(
      endString
    );


  var numberOfDays =
    endDate.difference(
      startDate,
      'day'
    );


  var dayOffsets =
    ee.List.sequence(
      0,
      numberOfDays.subtract(1)
    );


  var dailyFeatures =
    dayOffsets.map(
      function(offset) {

        var currentDay =
          startDate.advance(
            offset,
            'day'
          );


        var nextDay =
          currentDay.advance(
            1,
            'day'
          );


        // ==========================================================
        // 4A. HOURLY DATA FOR THIS UTC DAY
        // ==========================================================

        var hourlySub =
          era5HourlyProcessed.filterDate(
            currentDay,
            nextDay
          );


        var hourlyCount =
          hourlySub.size();


        // ==========================================================
        // 4B. TEMPORAL AGGREGATION
        // ==========================================================

        /*
         * These calculations happen across TIME.
         *
         * Example:
         *
         * 24 hourly VPD rasters
         *
         *      ↓
         *
         * one daily VPD raster
         */


        var t2mMean =
          hourlySub
            .select(
              't2m_C'
            )
            .mean();


        var dewpointMean =
          hourlySub
            .select(
              'td_C'
            )
            .mean();


        var rhMean =
          hourlySub
            .select(
              'rh_pct'
            )
            .mean();


        var vpdMean =
          hourlySub
            .select(
              'vpd_kPa'
            )
            .mean();


        /*
         * Daily VPD maximum:
         *
         * Each ERA5 pixel gets its own maximum
         * hourly VPD during the UTC day.
         *
         * Later we spatially average those
         * pixel-level daily maxima over AEZ-26.
         */

        var vpdMax =
          hourlySub
            .select(
              'vpd_kPa'
            )
            .max();


        var windMean =
          hourlySub
            .select(
              'wind_speed_m_s'
            )
            .mean();


        // ==========================================================
        // 4C. DAILY AGGREGATED ERA5-LAND IMAGE
        // ==========================================================

        var dailySub =
          era5DailyAggrCol.filterDate(
            currentDay,
            nextDay
          );


        var dailyCount =
          dailySub.size();


        var dailyImage =
          ee.Image(
            dailySub.first()
          );


        // Daily maximum and minimum temperature.
        var t2mMax =
          dailyImage
            .select(
              'temperature_2m_max'
            )
            .subtract(
              273.15
            );


        var t2mMin =
          dailyImage
            .select(
              'temperature_2m_min'
            )
            .subtract(
              273.15
            );


        // ==========================================================
        // 4D. RADIATION
        // ==========================================================

        /*
         * ERA5 radiation units:
         *
         * J/m² per day
         *
         * Convert to:
         *
         * MJ/m²/day
         *
         * divide by 1,000,000
         */


        var solarRadiation =
          dailyImage
            .select(
              'surface_solar_radiation_downwards_sum'
            )
            .divide(
              1e6
            );


        var netSolarRadiation =
          dailyImage
            .select(
              'surface_net_solar_radiation_sum'
            )
            .divide(
              1e6
            );


        var netThermalRadiation =
          dailyImage
            .select(
              'surface_net_thermal_radiation_sum'
            )
            .divide(
              1e6
            );


        /*
         * Net radiation =
         *
         * net shortwave
         * +
         * net longwave
         */

        var netRadiation =
          netSolarRadiation.add(
            netThermalRadiation
          );


        // ==========================================================
        // 4E. EVAPORATION
        // ==========================================================

        /*
         * ERA5 convention:
         *
         * downward vertical flux = positive
         *
         * evaporation normally = negative
         *
         *
         * Source unit:
         * meters water equivalent
         *
         * Convert to positive evaporation depth:
         *
         * × (-1000)
         *
         * -> mm/day
         *
         *
         * IMPORTANT:
         *
         * We do NOT silently clamp unusual values.
         *
         * GEE/ECMWF documentation warns that accumulated
         * flow bands can occasionally contain packing-related
         * negative or excessively high values.
         *
         * Those will be identified during CSV QC.
         */


        var totalEvaporation =
          dailyImage
            .select(
              'total_evaporation_sum'
            )
            .multiply(
              -1000
            );


        /*
         * This is ERA5 POTENTIAL EVAPORATION.
         *
         * It is NOT automatically equivalent
         * to FAO-56 reference ET0.
         */

        var potentialEvaporation =
          dailyImage
            .select(
              'potential_evaporation_sum'
            )
            .multiply(
              -1000
            );


        // ==========================================================
        // 4F. COMBINE DAILY RASTERS
        // ==========================================================

        var dailyMultiBand =
          ee.Image.cat([

            t2mMean.rename(
              't2m_mean_C'
            ),

            t2mMax.rename(
              't2m_max_C'
            ),

            t2mMin.rename(
              't2m_min_C'
            ),

            dewpointMean.rename(
              'dewpoint_mean_C'
            ),

            rhMean.rename(
              'rh_pct'
            ),

            vpdMean.rename(
              'vpd_mean_kPa'
            ),

            vpdMax.rename(
              'vpd_max_kPa'
            ),

            windMean.rename(
              'wind_speed_mean_m_s'
            ),

            solarRadiation.rename(
              'solar_radiation'
            ),

            netSolarRadiation.rename(
              'net_solar_radiation'
            ),

            netThermalRadiation.rename(
              'net_thermal_radiation'
            ),

            netRadiation.rename(
              'net_radiation'
            ),

            totalEvaporation.rename(
              'total_evaporation'
            ),

            potentialEvaporation.rename(
              'potential_evaporation'
            )

          ]);


        // ==========================================================
        // 4G. SPATIAL AGGREGATION
        // ==========================================================

        /*
         * Everything above converted:
         *
         * hourly data
         * ->
         * daily rasters
         *
         * That was TEMPORAL processing.
         *
         *
         * Now reduceRegion() converts:
         *
         * many ERA5 grid cells
         * ->
         * one AEZ-level value
         *
         * That is SPATIAL aggregation.
         */


        var stats =
          dailyMultiBand.reduceRegion({

            reducer:
              ee.Reducer.mean(),

            geometry:
              studyArea,

            scale:
              SCALE_ERA5,

            maxPixels:
              1e9,

            tileScale:
              2

          });


        // ==========================================================
        // 4H. ONE DAY -> ONE TABLE ROW
        // ==========================================================

        return ee.Feature(
          null,
          {

            date:
              currentDay.format(
                'YYYY-MM-dd'
              ),


            /*
             * QC metadata.
             *
             * Expected:
             *
             * hourly images = 24
             * daily aggregate images = 1
             */

            number_of_hourly_ERA5_images_used:
              hourlyCount,

            number_of_daily_ERA5_images_used:
              dailyCount,

            complete_day_flag:
              hourlyCount
                .eq(24)
                .and(
                  dailyCount.eq(1)
                ),


            // ---------------------------------------------
            // Temperature
            // ---------------------------------------------

            t2m_mean_C:
              stats.get(
                't2m_mean_C'
              ),

            t2m_max_C:
              stats.get(
                't2m_max_C'
              ),

            t2m_min_C:
              stats.get(
                't2m_min_C'
              ),

            dewpoint_mean_C:
              stats.get(
                'dewpoint_mean_C'
              ),


            // ---------------------------------------------
            // Atmospheric moisture demand
            // ---------------------------------------------

            relative_humidity_mean_pct:
              stats.get(
                'rh_pct'
              ),

            VPD_mean_kPa:
              stats.get(
                'vpd_mean_kPa'
              ),

            VPD_max_kPa:
              stats.get(
                'vpd_max_kPa'
              ),


            // ---------------------------------------------
            // Wind
            // ---------------------------------------------

            wind_speed_mean_m_s:
              stats.get(
                'wind_speed_mean_m_s'
              ),


            // ---------------------------------------------
            // Radiation
            // ---------------------------------------------

            solar_radiation_MJ_m2_day:
              stats.get(
                'solar_radiation'
              ),

            net_solar_radiation_MJ_m2_day:
              stats.get(
                'net_solar_radiation'
              ),

            net_thermal_radiation_MJ_m2_day:
              stats.get(
                'net_thermal_radiation'
              ),

            net_radiation_MJ_m2_day:
              stats.get(
                'net_radiation'
              ),


            // ---------------------------------------------
            // Evaporation
            // ---------------------------------------------

            total_evaporation_mm_day:
              stats.get(
                'total_evaporation'
              ),

            ERA5_potential_evaporation_mm_day:
              stats.get(
                'potential_evaporation'
              )

          }
        );

      }
    );


  return ee.FeatureCollection(
    dailyFeatures
  );

}
