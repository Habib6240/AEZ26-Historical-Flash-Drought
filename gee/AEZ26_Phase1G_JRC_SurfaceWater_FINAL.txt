 *
 * Study period:
 * 2000-01 to 2021-12
 *
 * Dataset:
 * JRC/GSW1_4/MonthlyHistory
 *
 * Native spatial resolution:
 * 30 m
 *
 * Native temporal resolution:
 * Monthly
 *
 * WATER CLASSES:
 *
 * 0 = No data
 * 1 = Not water
 * 2 = Water
 *
 * CRITICAL SCIENTIFIC RULES:
 *
 * 1. No observation != zero water.
 * 2. If no valid observation exists:
 *       water area = null
 *       water fraction = null
 * 3. If valid observations exist but no water is detected:
 *       water area = 0
 *       water fraction = 0
 * 4. Low spatial coverage is FLAGGED, not automatically deleted.
 * 5. JRC surface-water area is NOT surface-water storage volume.
 * 6. JRC km² cannot be directly subtracted from GRACE TWSA cm.
 *
 * FINAL OUTPUT:
 * AEZ26_JRC_SurfaceWater_Monthly_QC_2000_2021.csv
 * =====================================================================
 */


// =====================================================================
// 0. CONFIGURATION
// =====================================================================

var STUDY_AREA_ASSET =
  'projects/<YOUR-EE-PROJECT>/assets/<YOUR-AEZ26-ASSET>';


var DRIVE_FOLDER =
  'AEZ26_Final_Research_Data';


var START_DATE =
  '2000-01-01';


/*
 * filterDate end is exclusive.
 *
 * JRC v1.4 MonthlyHistory ends in December 2021.
 */

var END_DATE =
  '2022-01-01';


var SCALE_JRC =
  30;


/*
 * QC threshold.
 *
 * This does NOT remove observations.
 *
 * It only marks months where less than 70%
 * of the AEZ raster area has valid observations.
 */

var MIN_VALID_FRACTION =
  0.70;


// =====================================================================
// 1. STUDY AREA
// =====================================================================

var studyAreaFC =

  );


var studyArea =
  studyAreaFC.geometry();


var vectorAreaKm2 =
  studyArea

      maxError: 1

    .divide(
      1e6
    );

/*
 * Expected:
 *
 * 22 years × 12 months
 *
 * = 264 images.
 */


// =====================================================================
// 3. PROCESS ONE MONTH
// =====================================================================

function processJRC(
  image
) {


  // ===================================================================
  // 3A. NATIVE TIME / TRACEABILITY
  // ===================================================================

  var nativeDate =
    image.date();


  var nativeIndex =
    image.get(
      'system:index'
    );


  var year =
    ee.Number(
      nativeDate.get(
        'year'
      )
    );


  var month =
    ee.Number(
      nativeDate.get(
        'month'
      )
    );


  // ===================================================================
  // 3B. WATER CLASSIFICATION
  // ===================================================================

  var waterClass =
    image.select(
      'water'
    );


  /*
   * JRC classes:
   *
   * 0 = no data
   * 1 = valid observation, not water
   * 2 = valid observation, water
   */


  var validObservationMask =
    waterClass.gt(0);


  var waterMask =
    waterClass.eq(2);


  var nonWaterMask =
    waterClass.eq(1);


  // ===================================================================
  // 3C. PIXEL AREA
  // ===================================================================

  /*
   * ee.Image.pixelArea()
   *
   * area is initially m².
   *
   * Divide by 1,000,000:
   *
   * -> km²
   */


  var pixelAreaKm2 =
    ee.Image
      .pixelArea()
      .divide(
        1e6
      );


  // ===================================================================
  // 3D. REFERENCE RASTER AREA
  // ===================================================================

  /*
   * We use the JRC image mask to establish
   * the rasterized AEZ reference area.
   *
   * This is deliberately distinguished from
   * the geodesic vector area.
   */


  var referenceRasterAreaDict =
    pixelAreaKm2
      .updateMask(
        waterClass.mask()
      )
      .reduceRegion({

        reducer:
          ee.Reducer.sum(),

        geometry:
          studyArea,

        scale:
          SCALE_JRC,

        maxPixels:
          1e9,

        tileScale:
          4

      });


  var referenceRasterAreaKm2 =
    ee.Number(
      referenceRasterAreaDict.get(
        'area',
        0
      )
    );


  // ===================================================================
  // 3E. VALID OBSERVED AREA
  // ===================================================================

  /*
   * Only class 1 and class 2 are valid observations.
   *
   * Class 0 is NO DATA.
   */


  var validAreaDict =
    pixelAreaKm2
      .updateMask(
        validObservationMask
      )
      .reduceRegion({

        reducer:
          ee.Reducer.sum(),

        geometry:
          studyArea,

        scale:
          SCALE_JRC,

        maxPixels:
          1e9,

        tileScale:
          4

      });


  /*
   * If no valid pixels exist,
   * valid observed area is genuinely 0 km².
   *
   * This is QC coverage,
   * not water area.
   */


  var validAreaKm2 =
    ee.Number(
      validAreaDict.get(
        'area',
        0
      )
    );


  // ===================================================================
  // 3F. OBSERVED WATER AREA
  // ===================================================================

  var waterAreaDict =
    pixelAreaKm2
      .updateMask(
        waterMask
      )
      .reduceRegion({

        reducer:
          ee.Reducer.sum(),

        geometry:
          studyArea,

        scale:
          SCALE_JRC,

        maxPixels:
          1e9,

        tileScale:
          4

      });


  /*
   * If VALID observations exist but no water pixel exists,
   *
   * reduceRegion can return null.
   *
   * In that situation water area should genuinely be 0.
   */


  var observedWaterAreaKm2 =
    ee.Number(
      waterAreaDict.get(
        'area',
        0
      )
    );


  // ===================================================================
  // 3G. NON-WATER OBSERVED AREA
  // ===================================================================

  var nonWaterAreaDict =
    pixelAreaKm2
      .updateMask(
        nonWaterMask
      )
      .reduceRegion({

        reducer:
          ee.Reducer.sum(),

        geometry:
          studyArea,

        scale:
          SCALE_JRC,

        maxPixels:
          1e9,

        tileScale:
          4

      });


  var observedNonWaterAreaKm2 =
    ee.Number(
      nonWaterAreaDict.get(
        'area',
        0
      )
    );


  // ===================================================================
  // 3H. VALID OBSERVATION FRACTION
  // ===================================================================

  /*
   * Fraction of rasterized AEZ area
   * with valid water/non-water classification.
   */


  var validObservationFraction =
    ee.Algorithms.If(

      referenceRasterAreaKm2.gt(0),

      validAreaKm2.divide(
        referenceRasterAreaKm2
      ),

      null

    );


  // ===================================================================
  // 3I. FINAL WATER AREA
  // ===================================================================

  /*
   * SCIENTIFIC DISTINCTION:
   *
   * CASE A:
   *
   * validArea > 0,
   * no water detected
   *
   * -> water area = 0
   *
   *
   * CASE B:
   *
   * validArea = 0
   *
   * -> water area UNKNOWN
   * -> null
   */


  var finalWaterAreaKm2 =
    ee.Algorithms.If(

      validAreaKm2.gt(0),

      observedWaterAreaKm2,

      null

    );


  // ===================================================================
  // 3J. WATER FRACTION OF OBSERVED AREA
  // ===================================================================

  var waterFractionOfValidArea =
    ee.Algorithms.If(

      validAreaKm2.gt(0),

      observedWaterAreaKm2.divide(
        validAreaKm2
      ),

      null

    );


  // ===================================================================
  // 3K. NON-WATER FRACTION
  // ===================================================================

  var nonWaterFractionOfValidArea =
    ee.Algorithms.If(

      validAreaKm2.gt(0),

      observedNonWaterAreaKm2.divide(
        validAreaKm2
      ),

      null

    );


  // ===================================================================
  // 3L. QC FLAGS
  // ===================================================================

  var noValidObservationFlag =
    ee.Number(
      ee.Algorithms.If(

        validAreaKm2.gt(0),

        0,

        1

      )
    );


  var validFractionForFlag =
    ee.Number(
      ee.Algorithms.If(

        ee.Algorithms.IsEqual(
          validObservationFraction,
          null
        ),

        0,

        validObservationFraction

      )
    );


  var lowCoverageFlag =
    ee.Number(
      ee.Algorithms.If(

        validFractionForFlag.lt(
          MIN_VALID_FRACTION
        ),

        1,

        0

      )
    );


  /*
   * This indicates whether WATER was detected
   * among valid observations.
   *
   * It is NOT the same as data availability.
   */


  var waterDetectedFlag =
    ee.Number(
      ee.Algorithms.If(

        validAreaKm2.gt(0),

        ee.Algorithms.If(
          observedWaterAreaKm2.gt(0),
          1,
          0
        ),

        0

      )
    );


  // ===================================================================
  // 3M. INTERNAL AREA-BALANCE ERROR
  // ===================================================================

  /*
   * For valid observed pixels:
   *
   * water area
   * +
   * non-water area
   *
   * should approximately equal
   *
   * valid observed area.
   *
   * Tiny differences can arise from
   * floating-point / projection calculations.
   */


  var observedAreaBalanceErrorKm2 =
    observedWaterAreaKm2
      .add(
        observedNonWaterAreaKm2
      )
      .subtract(
        validAreaKm2
      )
      .abs();


  // ===================================================================
  // 3N. ONE MONTH -> ONE CSV ROW
  // ===================================================================

  return ee.Feature(
    null,
    {

      // ---------------------------------------------------------------
      // Native timing
      // ---------------------------------------------------------------

      native_system_index:
        nativeIndex,

      date:
        nativeDate.format(
          'YYYY-MM-dd'
        ),

      year:
        year,

      month:
        month,


      // ---------------------------------------------------------------
      // Reference areas
      // ---------------------------------------------------------------

      AEZ_vector_area_km2:
        vectorAreaKm2,

      JRC_reference_raster_area_km2:
        referenceRasterAreaKm2,


      // ---------------------------------------------------------------
      // Observation coverage
      // ---------------------------------------------------------------

      valid_observed_area_km2:
        validAreaKm2,

      valid_observation_fraction:
        validObservationFraction,


      // ---------------------------------------------------------------
      // Surface water
      // ---------------------------------------------------------------

      water_area_km2:
        finalWaterAreaKm2,

      water_fraction_of_valid_area:
        waterFractionOfValidArea,


      // ---------------------------------------------------------------
      // Non-water diagnostic
      // ---------------------------------------------------------------

      nonwater_observed_area_km2:
        observedNonWaterAreaKm2,

      nonwater_fraction_of_valid_area:
        nonWaterFractionOfValidArea,


      // ---------------------------------------------------------------
      // QC
      // ---------------------------------------------------------------

      no_valid_observation_flag:
        noValidObservationFlag,

      low_coverage_flag:
        lowCoverageFlag,

      water_detected_flag:
        waterDetectedFlag,

      observed_area_balance_error_km2:
        observedAreaBalanceErrorKm2

    }
  );

}
