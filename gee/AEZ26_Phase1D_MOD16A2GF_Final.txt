// =====================================================================
// 3. QA BIT HELPER
// =====================================================================

function extractBits(
  image,
  startBit,
  endBit
) {

  var bitMask = 0;

  for (
    var i = startBit;
    i <= endBit;
    i++
  ) {

    bitMask += Math.pow(
      2,
      i
    );

  }

  return image
    .bitwiseAnd(
      bitMask
    )
    .rightShift(
      startBit
    );
}


// =====================================================================
// 4. COVERAGE FRACTION HELPER
// =====================================================================

function fractionOverRegion(
  conditionImage
) {

  var binary =
    conditionImage
      .unmask(0)
      .rename(
        'fraction'
      );

  var result =
    binary.reduceRegion({

      reducer:
        ee.Reducer.mean(),

      geometry:
        studyArea,

      scale:
        SCALE_MOD16,

      maxPixels:
        1e9,

      tileScale:
        2

    });

  return ee.Number(
    result.get(
      'fraction'
    )
  );
}


// =====================================================================
// 5. SAFE RATIO HELPER
// =====================================================================

function safeRatio(
  numeratorValue,
  denominatorValue
) {

  return ee.Algorithms.If(

    ee.Algorithms.IsEqual(
      denominatorValue,
      null
    ),

    null,

    ee.Algorithms.If(

      ee.Algorithms.IsEqual(
        numeratorValue,
        null
      ),

      null,

      ee.Algorithms.If(

        ee.Number(
          denominatorValue
        ).gt(0),

        ee.Number(
          numeratorValue
        ).divide(
          ee.Number(
            denominatorValue
          )
        ),

        null

      )

    )

  );
}


// =====================================================================
// 6. SAFE PER-DAY NORMALIZATION
// =====================================================================

function safePerDay(
  value,
  numberOfDays
) {

  return ee.Algorithms.If(

    ee.Algorithms.IsEqual(
      value,
      null
    ),

    null,

    ee.Number(
      value
    ).divide(
      numberOfDays
    )

  );
}


// =====================================================================
// 7. PROCESS EACH MOD16 COMPOSITE
// =====================================================================

function processMOD16(
  image
) {


  // ===================================================================
  // 7A. NATIVE COMPOSITE TIMING
  // ===================================================================

  var compositeStart =
    image.date();

  var year =
    ee.Number(
      compositeStart.get(
        'year'
      )
    );

  var nextYear =
    ee.Date.fromYMD(
      year.add(1),
      1,
      1
    );

  var daysRemainingInYear =
    nextYear.difference(
      compositeStart,
      'day'
    );

  /*
   * Most composites = 8 days.
   * Final annual composite = 5 or 6 days.
   */

  var compositeDays =
    ee.Number(
      daysRemainingInYear
    ).min(
      8
    );

  var compositeEnd =
    compositeStart.advance(
      compositeDays.subtract(1),
      'day'
    );


  // ===================================================================
  // 7B. SOURCE BANDS
  // ===================================================================

  var rawET =
    image.select(
      'ET'
    );

  var rawPET =
    image.select(
      'PET'
    );

  var qc =
    image.select(
      'ET_QC'
    );


  // ===================================================================
  // 7C. PRIMARY VALID MASK
  // ===================================================================

  /*
   * Both ET and PET must exist.
   *
   * Additional valid stored-range protection:
   * 0–32700
   */

  var sourceMask =
    rawET
      .mask()
      .and(
        rawPET.mask()
      );

  var rangeMask =
    rawET
      .gte(0)
      .and(
        rawET.lte(32700)
      )
      .and(
        rawPET.gte(0)
      )
      .and(
        rawPET.lte(32700)
      );

  var primaryMask =
    sourceMask.and(
      rangeMask
    );


  // ===================================================================
  // 7D. SCALE ET / PET
  // ===================================================================

  /*
   * Scale factor = 0.1
   *
   * kg/m² water ≈ mm water depth.
   */

  var ETscaled =
    rawET
      .multiply(0.1)
      .rename(
        'ET_mm'
      );

  var PETscaled =
    rawPET
      .multiply(0.1)
      .rename(
        'PET_mm'
      );

  var ETprimary =
    ETscaled.updateMask(
      primaryMask
    );

  var PETprimary =
    PETscaled.updateMask(
      primaryMask
    );


  // ===================================================================
  // 7E. DECODE ET_QC
  // ===================================================================

  /*
   * Bit 0:
   * MODLAND QC
   *
   * 0 = good
   * 1 = other quality
   */

  var modlandQC =
    extractBits(
      qc,
      0,
      0
    );


  /*
   * Bits 3–4:
   * Cloud state
   *
   * 0 = clear
   * 1 = significant cloud
   * 2 = mixed cloud
   * 3 = undefined / assumed clear
   */

  var cloudState =
    extractBits(
      qc,
      3,
      4
    );


  /*
   * Bits 5–7:
   * SCF quality
   *
   * Strict sensitivity track keeps:
   * SCF <= 1
   */

  var scfQC =
    extractBits(
      qc,
      5,
      7
    );


  // ===================================================================
  // 7F. STRICT QA MASK
  // ===================================================================

  /*
   * This is deliberately strict.
   *
   * It is NOT used as the primary regional ET/PET series.
   */

  var strictMask =
    primaryMask

      .and(
        modlandQC.eq(0)
      )

      .and(
        cloudState.eq(0)
      )

      .and(
        scfQC.lte(1)
      );


  var ETstrict =
    ETscaled.updateMask(
      strictMask
    );

  var PETstrict =
    PETscaled.updateMask(
      strictMask
    );


  // ===================================================================
  // 7G. PRIMARY REGIONAL MEANS
  // ===================================================================

  var primaryStats =
    ee.Image.cat([

      ETprimary,

      PETprimary

    ]).reduceRegion({

      reducer:
        ee.Reducer.mean(),

      geometry:
        studyArea,

      scale:
        SCALE_MOD16,

      maxPixels:
        1e9,

      tileScale:
        2

    });


  var ETprimaryValue =
    primaryStats.get(
      'ET_mm'
    );

  var PETprimaryValue =
    primaryStats.get(
      'PET_mm'
    );


  // ===================================================================
  // 7H. STRICT QA REGIONAL MEANS
  // ===================================================================

  var strictStats =
    ee.Image.cat([

      ETstrict,

      PETstrict

    ]).reduceRegion({

      reducer:
        ee.Reducer.mean(),

      geometry:
        studyArea,

      scale:
        SCALE_MOD16,

      maxPixels:
        1e9,

      tileScale:
        2

    });


  var ETstrictValue =
    strictStats.get(
      'ET_mm'
    );

  var PETstrictValue =
    strictStats.get(
      'PET_mm'
    );


  // ===================================================================
  // 7I. COVERAGE FRACTIONS
  // ===================================================================

  var primaryFraction =
    fractionOverRegion(
      primaryMask
    );

  var strictFraction =
    fractionOverRegion(
      strictMask
    );

  var MODLANDgoodFraction =
    fractionOverRegion(

      primaryMask.and(
        modlandQC.eq(0)
      )

    );

  var clearPixelFraction =
    fractionOverRegion(

      primaryMask.and(
        cloudState.eq(0)
      )

    );

  var SCFgoodFraction =
    fractionOverRegion(

      primaryMask.and(
        scfQC.lte(1)
      )

    );


  // ===================================================================
  // 7J. ET/PET RATIOS
  // ===================================================================

  /*
   * Regional ratio:
   *
   * mean regional ET
   * ----------------
   * mean regional PET
   *
   * NOT mean(pixel ET/PET).
   */

  var primaryRatio =
    safeRatio(
      ETprimaryValue,
      PETprimaryValue
    );

  var strictRatio =
    safeRatio(
      ETstrictValue,
      PETstrictValue
    );


  // ===================================================================
  // 7K. NORMALIZED MM/DAY
  // ===================================================================

  /*
   * This does NOT create daily MODIS observations.
   *
   * It simply normalizes each composite total
   * by its actual number of days.
   */

  var ETprimaryPerDay =
    safePerDay(
      ETprimaryValue,
      compositeDays
    );

  var PETprimaryPerDay =
    safePerDay(
      PETprimaryValue,
      compositeDays
    );

  var ETstrictPerDay =
    safePerDay(
      ETstrictValue,
      compositeDays
    );

  var PETstrictPerDay =
    safePerDay(
      PETstrictValue,
      compositeDays
    );


  // ===================================================================
  // 7L. AVAILABILITY FLAGS
  // ===================================================================

  /*
   * Use explicit numeric 0 / 1.
   *
   * This avoids Boolean display/filter ambiguity
   * in Earth Engine Console.
   */

  var primaryAvailableFlag =
    ee.Number(
      ee.Algorithms.If(
        primaryFraction.gt(0),
        1,
        0
      )
    );

  var strictAvailableFlag =
    ee.Number(
      ee.Algorithms.If(
        strictFraction.gt(0),
        1,
        0
      )
    );


  // ===================================================================
  // 7M. ONE COMPOSITE -> ONE FEATURE
  // ===================================================================

  return ee.Feature(
    null,
    {

      // ---------------------------------------------------------------
      // TIME
      // ---------------------------------------------------------------

      composite_start_date:
        compositeStart.format(
          'YYYY-MM-dd'
        ),

      composite_end_date:
        compositeEnd.format(
          'YYYY-MM-dd'
        ),

      composite_days:
        compositeDays,


      // ---------------------------------------------------------------
      // PRIMARY GAP-FILLED TRACK
      // ---------------------------------------------------------------

      ET_primary_mm_composite:
        ETprimaryValue,

      PET_primary_mm_composite:
        PETprimaryValue,

      ET_primary_mm_day:
        ETprimaryPerDay,

      PET_primary_mm_day:
        PETprimaryPerDay,

      ET_PET_ratio_primary:
        primaryRatio,

      primary_valid_fraction:
        primaryFraction,

      primary_data_available_flag:
        primaryAvailableFlag,


      // ---------------------------------------------------------------
      // STRICT QA SENSITIVITY TRACK
      // ---------------------------------------------------------------

      ET_strictQA_mm_composite:
        ETstrictValue,

      PET_strictQA_mm_composite:
        PETstrictValue,

      ET_strictQA_mm_day:
        ETstrictPerDay,

      PET_strictQA_mm_day:
        PETstrictPerDay,

      ET_PET_ratio_strictQA:
        strictRatio,

      strict_QA_fraction:
        strictFraction,

      strict_QA_data_available_flag:
        strictAvailableFlag,


      // ---------------------------------------------------------------
      // QA COMPONENTS
      // ---------------------------------------------------------------

      MODLAND_good_fraction:
        MODLANDgoodFraction,

      clear_pixel_fraction:
        clearPixelFraction,

      SCF_0_or_1_fraction:
        SCFgoodFraction

    }
  );

}
