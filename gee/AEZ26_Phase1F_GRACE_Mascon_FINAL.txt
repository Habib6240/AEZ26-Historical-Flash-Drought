 *
 * IMPORTANT SCIENTIFIC RULES:
 *
 * 1. Preserve native GRACE observation timing.
 * 2. Do NOT interpolate missing months in GEE.
 * 3. Do NOT convert TWSA directly to groundwater.
 * 4. lwe_thickness = terrestrial water-storage anomaly
 *    in cm equivalent water thickness.
 * 5. uncertainty = provider 1-sigma mascon uncertainty.
 * 6. Regional mean uncertainty is NOT formal propagated
 *    uncertainty of the AEZ regional mean.
 * 7. 0.5-degree display/sampling grid does NOT mean
 *    independent 0.5-degree observations.
 *    Independent mascon information is much coarser (~3 degrees).
 *
 * OUTPUT:
 * AEZ26_GRACE_Mascon_TWSA_Uncertainty.csv
 * =====================================================================
 */

// =====================================================================
// 4. PROCESS ONE NATIVE GRACE OBSERVATION
// =====================================================================

function processGRACE(
  image
) {


  // ===================================================================
  // 4A. NATIVE TRACEABILITY
  // ===================================================================

  var nativeIndex =
    image.get(
      'system:index'
    );


  var nativeTimestampMillis =
    image.get(
      'system:time_start'
    );


  var nativeDate =
    image
      .date()
      .format(
        'YYYY-MM-dd'
      );


  var nativeTimestampUTC =
    image
      .date()
      .format(
        'YYYY-MM-dd HH:mm:ss'
      );


  /*
   * Optional native end date.
   *
   * We check whether the property exists.
   * If absent, result remains null.
   */

  var hasTimeEnd =
    image
      .propertyNames()
      .contains(
        'system:time_end'
      );


  var nativeEndDate =
    ee.Algorithms.If(

      hasTimeEnd,

      ee.Date(
        image.get(
          'system:time_end'
        )
      ).format(
        'YYYY-MM-dd'
      ),

      null

    );


  // ===================================================================
  // 4B. SOURCE BANDS
  // ===================================================================

  /*
   * lwe_thickness:
   *
   * equivalent liquid-water thickness
   *
   * unit:
   * cm
   *
   * Interpretation:
   * terrestrial water-storage anomaly
   *
   * NOT groundwater alone.
   */


  var twsa =
    image.select(
      'lwe_thickness'
    );


  /*
   * uncertainty:
   *
   * provider 1-sigma uncertainty
   * associated with mascon estimate.
   */

  var uncertainty =
    image.select(
      'uncertainty'
    );


  // ===================================================================
  // 4C. REGIONAL SPATIAL MEANS
  // ===================================================================

  /*
   * We calculate AEZ-level raster-grid spatial means.
   *
   * IMPORTANT:
   *
   * Adjacent ~0.5-degree output pixels are NOT
   * independent mascon observations.
   */


  var twsaStats =
    twsa.reduceRegion({

      reducer:
        ee.Reducer.mean(),

      geometry:
        studyArea,

      scale:
        SCALE_GRACE,

      maxPixels:
        1e9

    });


  var uncertaintyStats =
    uncertainty.reduceRegion({

      reducer:
        ee.Reducer.mean(),

      geometry:
        studyArea,

      scale:
        SCALE_GRACE,

      maxPixels:
        1e9

    });


  var twsaValue =
    twsaStats.get(
      'lwe_thickness'
    );


  var uncertaintyValue =
    uncertaintyStats.get(
      'uncertainty'
    );


  // ===================================================================
  // 4D. DATA AVAILABILITY
  // ===================================================================

  var dataAvailable =
    ee.Number(
      ee.Algorithms.If(

        ee.Algorithms.IsEqual(
          twsaValue,
          null
        ),

        0,

        1

      )
    );


  // ===================================================================
  // 4E. ONE NATIVE OBSERVATION -> ONE CSV ROW
  // ===================================================================

  return ee.Feature(
    null,
    {

      native_system_index:
        nativeIndex,

      native_timestamp_millis:
        nativeTimestampMillis,

      native_timestamp_UTC:
        nativeTimestampUTC,

      native_date:
        nativeDate,

      native_end_date_if_available:
        nativeEndDate,


      /*
       * Main research variable.
       */

      TWSA_cm:
        twsaValue,


      /*
       * IMPORTANT NAME:
       *
       * This is the spatial mean of provider
       * mascon uncertainty values over AEZ-26.
       *
       * It is NOT a formal propagated uncertainty
       * of the regional mean TWSA.
       */

      mean_mascon_uncertainty_cm:
        uncertaintyValue,


      TWSA_data_available_flag:
        dataAvailable

    }
  );

}
